/* ---------- Loader ---------- */
var loaderDone=false;
function finishLoader(){
  if(loaderDone) return; loaderDone=true;
  var l=document.getElementById('loader'); if(l) l.classList.add('done');
  var solo=document.getElementById('mvSolo'); if(solo) solo.classList.add('in');
}
window.addEventListener('load', function(){ setTimeout(finishLoader, 900); });

/* ---------- Hero slideshow（1人ずつ・3秒で切替） ---------- */
(function(){
  var root = document.getElementById('mvSolo');
  if(!root) return;
  var slides = root.querySelectorAll('.mv-slide');
  var names  = root.querySelectorAll('.mv-name');
  var dots   = root.querySelectorAll('.mv-dot');
  if(slides.length < 2) return;

  var INTERVAL = 3000;
  var cur = 0, timer = null, paused = false;

  function setActive(list, n){
    Array.prototype.forEach.call(list, function(el, i){
      var on = (i === n);
      el.classList.toggle('is-active', on);
      if(el.tagName === 'A'){
        el.setAttribute('tabindex', on ? '0' : '-1');
        el.setAttribute('aria-hidden', on ? 'false' : 'true');
      }
    });
  }
  function show(n){
    n = (n + slides.length) % slides.length;
    var prev = slides[cur];
    if(prev && n !== cur){
      prev.classList.add('is-out');
      void prev.offsetWidth;
      prev.classList.add('is-out-go');
      (function(el){ setTimeout(function(){ el.classList.remove('is-out','is-out-go'); }, 1000); })(prev);
    }
    cur = n;
    setActive(slides, n);
    setActive(names, n);
    /* バーのアニメーションを毎回リスタートさせる */
    Array.prototype.forEach.call(dots, function(d){ d.classList.remove('is-active'); });
    var d = dots[n];
    if(d){ void d.offsetWidth; d.classList.add('is-active'); }
  }
  function start(){ stop(); if(!paused) timer = setInterval(function(){ show(cur + 1); }, INTERVAL); }
  function stop(){ if(timer){ clearInterval(timer); timer = null; } }

  Array.prototype.forEach.call(dots, function(d){
    d.addEventListener('click', function(){
      var n = parseInt(d.getAttribute('data-i'), 10) || 0;
      if(n === cur) return;
      show(n); start();
    });
  });

  /* マウスが乗っている間だけ止める（指で操作する端末では止めない） */
  if(window.matchMedia && window.matchMedia('(hover:hover)').matches){
    root.addEventListener('mouseenter', function(){ paused = true; root.classList.add('is-paused'); stop(); });
    root.addEventListener('mouseleave', function(){ paused = false; root.classList.remove('is-paused'); start(); });
  }
  /* 別タブに行っている間は進めない */
  document.addEventListener('visibilitychange', function(){
    if(document.hidden){ stop(); } else if(!paused){ show(cur); start(); }
  });

  /* 2枚目以降を先読みして、切り替わりの瞬間に白く抜けないようにする */
  window.addEventListener('load', function(){
    Array.prototype.forEach.call(slides, function(sl){
      var im = sl.querySelector('img');
      if(im){ im.setAttribute('loading','eager'); im.setAttribute('decoding','sync'); }
    });
  });

  show(0);
  start();
})();

var hdr=document.getElementById('hdr'), lastY=0;
window.addEventListener('scroll', function(){
  var y=window.scrollY;
  if(hdr){
    hdr.classList.toggle('solid', y>40);
    if(y>lastY && y>300 && !document.body.classList.contains('menu-open')){ hdr.classList.add('hide'); }
    else { hdr.classList.remove('hide'); }
  }
  lastY=y;
}, {passive:true});

/* ---------- Mobile menu ---------- */
var bg=document.getElementById('burger');
if(bg) bg.addEventListener('click', function(){ document.body.classList.toggle('menu-open'); });
Array.prototype.forEach.call(document.querySelectorAll('#mmenu a'), function(a){
  a.addEventListener('click', function(){ document.body.classList.remove('menu-open'); });
});

/* ---------- Scroll reveal ---------- */
if('IntersectionObserver' in window){
  var io=new IntersectionObserver(function(es){
    es.forEach(function(e){ if(e.isIntersecting){ e.target.classList.add('in'); io.unobserve(e.target); } });
  },{threshold:.12,rootMargin:'0px 0px -6% 0px'});
  Array.prototype.forEach.call(document.querySelectorAll('.rv'), function(el){ io.observe(el); });
} else {
  Array.prototype.forEach.call(document.querySelectorAll('.rv'), function(el){ el.classList.add('in'); });
}
</script>
